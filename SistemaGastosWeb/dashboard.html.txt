    <script>
        const API_URL = "https://localhost:7031/api"; // TU PUERTO

        const token = localStorage.getItem('token');
        if (!token) window.location.href = 'index.html';

        let chartInstance = null;

        document.addEventListener('DOMContentLoaded', () => {
            const now = new Date();
            now.setMinutes(now.getMinutes() - now.getTimezoneOffset());
            document.getElementById('txtFecha').value = now.toISOString().slice(0, 16);
            cargarDatosDashboard();
            cargarPresupuestos();
            cargarListasDesplegables();
        });

        async function cargarDatosDashboard() {
            try {
                const response = await fetch(`${API_URL}/Gastos`, { headers: { 'Authorization': `Bearer ${token}` } });
                if (response.status === 401) throw new Error("Unauthorized");
                const gastos = await response.json();

                document.getElementById('totalCount').innerText = gastos.length;
                const total = gastos.reduce((sum, g) => sum + g.monto, 0);
                document.getElementById('totalAmount').innerText = `$${total.toLocaleString('en-US', {minimumFractionDigits: 2})}`;
                actualizarTablaYGrafica(gastos);
            } catch (error) { if(error.message === "Unauthorized") logout(); }
        }

        async function cargarPresupuestos() {
            try {
                const date = new Date();
                const anio = date.getFullYear();
                const mes = date.getMonth() + 1;
                const response = await fetch(`${API_URL}/Presupuestos/reporte?anio=${anio}&mes=${mes}`, { headers: { 'Authorization': `Bearer ${token}` } });
                if (!response.ok) return;
                const reporte = await response.json();
                
                const container = document.getElementById('listaPresupuestos');
                container.innerHTML = '';
                if (reporte.length === 0) container.innerHTML = '<p class="text-muted small">No hay límites definidos.</p>';

                reporte.forEach(item => {
                    let colorClass = 'bg-success';
                    if (item.colorAlerta === 'Amarillo') colorClass = 'bg-warning';
                    if (item.colorAlerta === 'Naranja') colorClass = 'bg-warning';
                    if (item.colorAlerta === 'Rojo') colorClass = 'bg-danger';
                    let ancho = item.porcentajeConsumido > 100 ? 100 : item.porcentajeConsumido;

                    container.innerHTML += `
                        <div class="mb-3">
                            <div class="d-flex justify-content-between small mb-1"><span class="fw-bold text-dark">${item.categoria}</span><span class="text-muted">$${item.totalGastado} / $${item.limitePresupuesto}</span></div>
                            <div class="progress" style="height: 8px;"><div class="progress-bar ${colorClass}" style="width: ${ancho}%"></div></div>
                            <small class="${colorClass.replace('bg-', 'text-')} fw-bold" style="font-size: 0.7rem;">${item.mensajeAlerta} (${item.porcentajeConsumido}%)</small>
                        </div>`;
                });
            } catch (error) { console.error(error); }
        }

        function actualizarTablaYGrafica(gastos) {
            const tbody = document.getElementById('tablaGastos');
            tbody.innerHTML = '';
            const categoriasMap = {};

            gastos.forEach(g => {
                tbody.innerHTML += `<tr><td class="text-muted small">${new Date(g.fecha).toLocaleDateString()}</td><td class="fw-bold text-dark">${g.descripcion}</td><td><span class="badge-cat">${g.categoria}</span></td><td class="text-end fw-bold text-dark">$${g.monto.toFixed(2)}</td></tr>`;
                if (categoriasMap[g.categoria]) categoriasMap[g.categoria] += g.monto; else categoriasMap[g.categoria] = g.monto;
            });

            const ctx = document.getElementById('gastosChart').getContext('2d');
            if (chartInstance) chartInstance.destroy();
            chartInstance = new Chart(ctx, {
                type: 'doughnut',
                data: { labels: Object.keys(categoriasMap), datasets: [{ data: Object.values(categoriasMap), backgroundColor: ['#23a6d5', '#e73c7e', '#23d5ab', '#ee7752', '#6f42c1'], borderWidth: 0 }] },
                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom', labels: { usePointStyle: true } } } }
            });
        }

        async function cargarListasDesplegables() {
            try {
                const [resCat, resMet] = await Promise.all([fetch(`${API_URL}/Categorias`, { headers: { 'Authorization': `Bearer ${token}` } }), fetch(`${API_URL}/MetodosPago`, { headers: { 'Authorization': `Bearer ${token}` } })]);
                const cats = await resCat.json(); const mets = await resMet.json();
                const selCat = document.getElementById('selCategoria'); const selMet = document.getElementById('selMetodo'); const selPre = document.getElementById('selCategoriaPresup');
                selCat.innerHTML = ''; selMet.innerHTML = ''; selPre.innerHTML = '';
                cats.forEach(c => { const opt = `<option value="${c.id}">${c.nombre}</option>`; selCat.innerHTML += opt; selPre.innerHTML += opt; });
                mets.forEach(m => selMet.innerHTML += `<option value="${m.id}">${m.nombre}</option>`);
            } catch (error) { console.error(error); }
        }

        // --- MANEJO DE ENVIO
        document.getElementById('formGasto').addEventListener('submit', async (e) => { e.preventDefault(); const d = { descripcion: document.getElementById('txtDescripcion').value, monto: parseFloat(document.getElementById('txtMonto').value), fecha: document.getElementById('txtFecha').value, categoriaId: parseInt(document.getElementById('selCategoria').value), metodoPagoId: parseInt(document.getElementById('selMetodo').value) }; await enviarDatos(`${API_URL}/Gastos`, d, 'modalGasto', true); });
        document.getElementById('formCategoria').addEventListener('submit', async (e) => { e.preventDefault(); await enviarDatos(`${API_URL}/Categorias`, { nombre: document.getElementById('txtNombreCat').value }, 'modalCategoria', false); cargarListasDesplegables(); document.getElementById('txtNombreCat').value = ''; });
        document.getElementById('formMetodo').addEventListener('submit', async (e) => { e.preventDefault(); await enviarDatos(`${API_URL}/MetodosPago`, { nombre: document.getElementById('txtNombreMet').value, icono: "fas fa-wallet" }, 'modalMetodo', false); cargarListasDesplegables(); document.getElementById('txtNombreMet').value = ''; });
        document.getElementById('formPresupuesto').addEventListener('submit', async (e) => { e.preventDefault(); const date = new Date(); await enviarDatos(`${API_URL}/Presupuestos`, { categoriaId: parseInt(document.getElementById('selCategoriaPresup').value), montoMaximo: parseFloat(document.getElementById('txtMontoPresup').value), anio: date.getFullYear(), mes: date.getMonth() + 1 }, 'modalPresupuesto', true); });

        async function enviarDatos(url, data, modalId, recargar) {
            try {
                const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` }, body: JSON.stringify(data) });
                if (response.ok) {
                    bootstrap.Modal.getInstance(document.getElementById(modalId)).hide();
                    if (recargar) { document.getElementById('formGasto').reset(); cargarDatosDashboard(); cargarPresupuestos(); }
                    alert("¡Guardado exitosamente!");
                } else { const err = await response.json(); console.log(err); alert("Error: " + (err.error || JSON.stringify(err))); }
            } catch (error) { alert("Error de conexión"); }
        }

        async function subirArchivo(input) {
            if (!input.files || input.files.length === 0) return;
            const formData = new FormData(); formData.append('archivo', input.files[0]);
            try {
                alert("Subiendo archivo...");
                const response = await fetch(`${API_URL}/Gastos/importar`, { method: 'POST', headers: { 'Authorization': `Bearer ${token}` }, body: formData });
                const res = await response.json();
                if (response.ok) { alert(res.mensaje); cargarDatosDashboard(); } else { alert("Error: " + (res.error || "Fallo en la importación")); }
            } catch (error) { console.error(error); alert("Error de conexión"); }
            input.value = '';
        }

        function logout() { localStorage.removeItem('token'); window.location.href = 'index.html'; }
        
        async function exportar() {
            try {
                const response = await fetch(`${API_URL}/Gastos/exportar`, { method: 'GET', headers: { 'Authorization': `Bearer ${token}` } });
                if (response.ok) { const blob = await response.blob(); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = "MisGastos.csv"; document.body.appendChild(a); a.click(); a.remove(); window.URL.revokeObjectURL(url); } else { alert("Error descarga"); }
            } catch (error) { alert("Error conexión"); }
        }
    </script>
